\begin{abstract}
[\blue{Blue highlights the research questions and reading guide;} 
\red{red highlights the main answers and limitations.}]

In this draft, we consider the problem of finding a fixed point of a nonexpansive operator
in $\RR^d$ from stochastic oracle evaluations. Many single-loop stochastic
first-order methods, such as stochastic gradient descent and stochastic
splitting methods, are stochastic Krasnosel'ski\u{i}--Mann (KM) iterations
\cite{combettes2015stochastic,bravo2024stochastic}. For exact evaluations,
the Fast KM method of Bo\c{t} and Nguyen \cite{bot2023fast} improves the
last-iterate residual of KM from $o(1/\sqrt{k})$ to $o(1/k)$ through
momentum. Under bounded variance, the last iterate of stochastic KM attains
only $\log k/k^{1/6}$ \cite{bravo2024stochastic}. \blue{We therefore study whether the
momentum of Fast KM accelerates the last iterate when the operator values
are noisy.}

\end{abstract}

\noindent\textbf{Key words.} nonexpansive operator, fixed-point problem,
Fast KM, stochastic oracle, almost-sure convergence.

\section{Introduction}\label{sec:introduction}
In this draft, we consider the problem of finding a fixed point of a nonexpansive operator
$T:\HH\to\HH$ when only noisy evaluations of $T$
are available. Many single-loop stochastic first-order methods are
stochastic Krasnosel'ski\u{i}--Mann (KM) iterations of the form
\begin{equation}\label{eq:intro-skm}
 x_{k+1}=(1-\lambda_k)x_k+\lambda_k\bigl(Tx_k+e_k\bigr),
\end{equation}
where $e_k$ is the error of the $k$th oracle response
\cite{combettes2015stochastic,bravo2024stochastic,cortild2026stochastic}.
Stochastic gradient descent, the stochastic proximal gradient method and
stochastic backward--forward splitting are instances of this form; in the
proximal gradient method, the error is in general conditionally biased.

For exact evaluations, the fixed-point residual of KM with a constant
relaxation parameter satisfies $\norm{x_k-Tx_k}=o(1/\sqrt{k})$
\cite{bot2023fast}. The Fast KM method of Bo\c{t} and Nguyen
\cite[Algorithm~2.1]{bot2023fast} improves this to
\begin{equation}\label{eq:intro-fast-km-rate}
 \norm{x_k-Tx_k}=o(1/k),\qquad x_k\to x^\star\in\Fix T,
\end{equation}
for a momentum parameter $\alpha>2$. 
% Each Fast KM update is a KM step
% followed by an extrapolation of operator values
% \cite[Remark~2.1]{bot2023fast}; it uses $Tx_k$ and the value $Tx_{k-1}$
% computed in the preceding step (Section~\ref{sec:deterministic-fast-km}). Applied to the forward--backward
% map, Fast KM yields an accelerated $o(1/k)$ rate for the proximal
% gradient method \cite[Corollary~4.1]{bot2023fast}.
% Table~\ref{tab:rates} compares the known last-iterate rates.

With noisy evaluations, $Tx_k$ in Fast KM is replaced
by an oracle response $\widehat T_k(x_k)=Tx_k+e_k$. For the value at
$x_{k-1}$ there are two implementations.
\begin{itemize}
 \item \emph{Re-query with the same sample.} The sampled operator of
 iteration $k$ is evaluated at $x_k$ and at $x_{k-1}$. This uses two
 queries per iteration and an oracle that can evaluate one sample at two
 points \cite[Assumption~2.3]{lee2026fixed}. Variance-reduced methods
 build on this access \cite{trandinh2024vrfkm,trandinh2025vfosa,lee2026fixed}.
 \item \emph{Reuse.} The response
 $\widehat T_{k-1}(x_{k-1})=Tx_{k-1}+e_{k-1}$ of the preceding step is
 stored and used again. This keeps one query per iteration, as in the
 deterministic method, and needs only fresh samples.
\end{itemize}
\blue{We thus study the reuse implementation.} Table~\ref{tab:rates} raises three questions:
\blue{
\begin{enumerate}
 \item \emph{(Q1) Last iterate of stochastic KM.} Under bounded variance, the best
 available last-iterate bound is $O(\log k/k^{1/6})$, whereas a randomly
 selected iterate attains $O(k^{-1/4})$. Does the last iterate necessarily
 lose this rate? (Section~\ref{sec:km})
 \item \emph{(Q2) Implicit variance reduction in Fast KM.} Can the momentum of Fast KM reduce the variance
 implicitly and improve $O(\log k/k^{1/6})$ under bounded variance?
 (Section~\ref{sec:algorithm})
 \item \emph{(Q3) Fast KM under decaying noise.} Under bounded variance, the residual
 after $k$ samples is $\Omega(\sigma/\sqrt{k})$ on the whole problem class
 \cite[Theorem~2]{foster2019complexity}. How fast must the variance decay
 for the reuse implementation to attain $o(1/k)$?
 (Section~\ref{sec:decaying})
\end{enumerate}}

\section{Preliminaries}\label{sec:problem}

The problem is
\begin{equation}\label{eq:problem}
 \text{find }x^\star\in\HH\quad\text{such that}\quad Tx^\star=x^\star,
 \tag{FP}
\end{equation}
where $T$ satisfies the following assumption.
\begin{assumption}[Operator]\label{ass:operator}
$T:\HH\to\HH$ is nonexpansive, that is, $\norm{Tx-Ty}\leq\norm{x-y}$ for
all $x,y\in\HH$, and $\Fix T:=\{x\in\HH:Tx=x\}$ is nonempty.
\end{assumption}
% The iterate accuracy is measured by the residual $\norm{x-Tx}$. With $V:=\Id-T$, nonexpansiveness of $T$ is equivalent to
% \begin{equation}\label{eq:cocoercivity}
%  \ip{Vx-Vy}{x-y}\geq\frac12\norm{Vx-Vy}^{2},\qquad x,y\in\HH.
% \end{equation}

Let $(\Omega,\FF,\PP)$ be a probability space and $(\FF_k)_{k\geq0}$ a
filtration, where $\FF_k$ contains the information available before query
$k$. At an $\FF_k$-measurable point $x_k$, the oracle returns
\begin{equation}\label{eq:oracle}
 \widehat T_k(x_k)=Tx_k+e_k
\end{equation}
with an $\FF_{k+1}$-measurable error $e_k$.

\begin{assumption}[Oracle]\label{ass:oracle}
For every $k\geq0$,
\begin{equation}\label{eq:h1}
 \EE[e_k\mid\FF_k]=0\qquad\text{and}\qquad
 \sigma_k^2:=\EE\bigl[\norm{e_k}^2\bigm|\FF_k\bigr]
 \ \text{with}\ \EE\sigma_k^2<\infty.
 \tag{H1}
\end{equation}
\end{assumption}

\section{The last iterate of stochastic KM}\label{sec:km}
\blue{In this section, we review existing convergence results for stochastic
Krasnosel'ski\u{i}--Mann iterations.}

For Problem~\eqref{eq:problem}, the Krasnosel'ski\u{i}--Mann (KM)
iteration reads
\begin{equation}\label{eq:km}
 x_{k+1}=(1-\lambda_{k+1})x_k+\lambda_{k+1}Tx_k,\qquad k\geq0,
\end{equation}
with $x_0\in\HH$ and parameters $\lambda_k\in(0,1)$. 

With a constant $\lambda_k\equiv\lambda$, the iterates converge to a point of $\Fix T$, and the rate is
\[
\norm{x_k-Tx_k}=o(1/\sqrt{k}).
\]

The stochastic KM iteration is
\begin{equation}\label{eq:skm}
 x_{k+1}=(1-\lambda_{k+1})x_k+\lambda_{k+1}\widehat T_k(x_k),\qquad
 \widehat T_k(x_k)=Tx_k+e_k .
\end{equation} under bounded variance, a randomly selected iterate generated by \eqref{eq:skm} attains the order $k^{-1/4}$ \cite[Corollary~5.4]{bravo2024stochastic}, \cite{cortild2026stochastic}.  This naturally introduce the following question:
\blue{
\begin{quote}
\emph{(Q1) For a general nonexpansive $T$ and bounded variance, can the
last iterate of stochastic KM attain the order $k^{-1/4}$?}
\end{quote}}

We recall the last-iterate bound of \cite{bravo2024stochastic}.   Let $D:=\dist(x_0,\Fix T)$ and set
$\tau_k:=\sum_{j=1}^{k}\lambda_j(1-\lambda_j)$,
$\varphi(y):=\min\{1,1/\sqrt{\pi y}\}$ and
\[
 \nu_k^2:=\sum_{j=1}^{k}\bigl(w_j^k\bigr)^2\,\EE\sigma_{j-1}^2,
 \qquad w_j^k:=\lambda_j\prod_{i=j+1}^{k}(1-\lambda_i),
\]
so that $\tau_k$ is the effective time of the iteration and $\nu_k$ the
scale of the averaged noise. 

\begin{assumption}\label{ass:h4}
Either $T$ has bounded range and $\bar\kappa\geq\sup_{x}\norm{Tx-x_0}$, or
$S:=\sum_{j\geq2}\lambda_j\nu_{j-1}<\infty$ and $\bar\kappa\geq2D+S$.
\end{assumption}

\begin{theorem}[{Last-iterate residual of stochastic KM \cite[Theorem~2.10]{bravo2024stochastic}}]\label{thm:skm}
Let Assumptions~\ref{ass:operator}, \ref{ass:oracle} and~\ref{ass:h4} hold.
Then, for every $k\geq1$,
\begin{equation}\label{eq:skm-bound}
 \EE\norm{x_k-Tx_k}\leq\bar\kappa\,\varphi(\tau_k)
 +2\sum_{j=2}^{k}\lambda_j\,\varphi(\tau_k-\tau_j)\,\nu_{j-1}
 +4\nu_k .
\end{equation}
\end{theorem}

We now consider the special case of Assumption~\ref{ass:oracle} in which
the second moments are uniformly bounded.

\begin{assumption}[Bounded variance]\label{ass:bvar}
There exists a finite constant $\sigma\geq0$ such that
$\sup_{k\geq0}\EE\sigma_k^2\leq\sigma^2$.
\end{assumption}

\begin{corollary}{\cite[Theorem~4.4]{bravo2024stochastic}}\label{cor:skm}
Under the assumptions of Theorem~\ref{thm:skm} and
Assumption~\ref{ass:bvar}, let $\lambda_k=(k+1)^{-a}$ with
$a\in(\frac12,1)$. Then $\tau_k\asymp k^{1-a}$ and it holds that
\[
 \EE\norm{x_k-Tx_k}\lesssim
 \underbrace{\frac{1}{\sqrt{\tau_k}}}_{\text{deterministic}}
 +\underbrace{\sigma\sqrt{k}\,\lambda_k}_{\text{noise}}
 =\frac{1}{k^{(1-a)/2}}+\frac{\sigma}{k^{a-1/2}}.
\]
Moreover, it holds that 
$\EE\norm{x_k-Tx_k}=O(\log k/k^{1/6})$ with $a=\frac23$.
\end{corollary}

\paragraph{\red{Answer to (Q1).}}\label{par:answer-q1}
\red{In the literature examined here, No: for a general nonexpansive $T$,
\[
 \EE\norm{x_k-Tx_k}=O\Bigl(\frac{\log k}{k^{1/6}}\Bigr)
 \quad\text{(last iterate)},\qquad
 \EE\norm{x_R-Tx_R}=O\Bigl(\frac{1}{k^{1/4}}\Bigr)
 \quad\text{(random iterate)}.
\]}
\red{The limitations are }
\begin{enumerate}
 \item  For exact evaluations,
 \[
  \norm{x_{k+1}-Tx_{k+1}}\leq\norm{x_k-Tx_k}
  \quad\Longrightarrow\quad
  \norm{x_k-Tx_k}^2\leq\frac{D^2}{\tau_{k+1}} ;
 \]
 with noise, $\EE\norm{x_k-Tx_k}^2$ need not be monotone, and the summed
 bound reaches only $x_R$ \cite[Remark~5.3]{bravo2024stochastic}.
 \item \emph{Order of the last iterate:} in~\eqref{eq:skm-bound},
 $1/\sqrt{\tau_k}\asymp\sqrt{k}\,\lambda_k\asymp k^{-1/6}$ at $a=\frac23$
 (Corollary~\ref{cor:skm}).
\end{enumerate}

\section{Implicit variance reduction in Fast KM}\label{sec:algorithm}
\subsection{Fast KM iteration}\label{sec:deterministic-fast-km}

Bo\c{t} and Nguyen \cite{bot2023fast} accelerate the KM iteration using
Nesterov's momentum updates. For fixed parameters $\alpha>2$ and
$0<s\leq1$, and initial points $x_0,x_1\in\HH$, their Fast KM method is
\begin{equation}\label{eq:deterministic-fast-km}
\begin{split}
 x_{k+1}={}&\left(1-\frac{s\alpha}{2(k+\alpha)}\right)x_k
 +\frac{(1-s)k}{k+\alpha}(x_k-x_{k-1})
 +\frac{s\alpha}{2(k+\alpha)}Tx_k\\
 &+{\frac{sk}{k+\alpha}(Tx_k-Tx_{k-1})},\qquad k\geq1.
\end{split}
\end{equation}

The stepsize $s$ can be absorbed into the operator: \eqref{eq:deterministic-fast-km}
applied to $T$ coincides with \eqref{eq:deterministic-fast-km} with $s=1$
applied to $M:=(1-s)\Id+sT$, which is nonexpansive with $\Fix M=\Fix T$,
and an oracle for $T$ with error $e_k$ yields an oracle for $M$ with error
$se_k$. We therefore take $s=1$ without loss of generality: 
\begin{equation}\label{eq:fast-km-unit-step}
 x_{k+1}=\left(1-\frac{\alpha}{2(k+\alpha)}\right)x_k
 +\frac{\alpha}{2(k+\alpha)}Tx_k
 +\frac{k}{k+\alpha}(Tx_k-Tx_{k-1}).
\end{equation}

\subsection{The reuse implementation of Stochastic Fast KM}\label{sec:iteration}

With noisy evaluations, $Tx_k$ in~\eqref{eq:fast-km-unit-step} is replaced
by a fresh response $\widehat T_k(x_k)=Tx_k+e_k$, while $Tx_{k-1}$ is not
re-queried: we reuse the response stored at the previous step,
\[
 \widehat T_{k-1}(x_{k-1})=Tx_{k-1}+e_{k-1},
\]
where $e_{k-1}$ is the error already realized at step $k-1$. Each iteration
therefore calls the oracle exactly once, and the update reads
\[
 x_{k+1}=x_k-\frac{\alpha}{2(k+\alpha)}\bigl(x_k-\widehat T_k(x_k)\bigr)
 +\frac{k}{k+\alpha}\bigl(\widehat T_k(x_k)-\widehat T_{k-1}(x_{k-1})\bigr).
\]
\begin{algorithm}[t]
\caption{Stochastic Fast KM with reuse}\label{alg:sfkm}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $x_0,x_1\in\HH$, $\alpha>2$.
\STATE Query $\widehat T_0(x_0)$ and store it.
\FOR{$k=1,2,\dots$}
 \STATE Query $\widehat T_k(x_k)=Tx_k+e_k$.
 \STATE $x_{k+1}=x_k-\frac{\alpha}{2(k+\alpha)}\bigl(x_k-\widehat T_k(x_k)\bigr)
 +\frac{k}{k+\alpha}\bigl(\widehat T_k(x_k)-\widehat T_{k-1}(x_{k-1})\bigr)$.
 \STATE Store $\widehat T_k(x_k)$.
\ENDFOR
\end{algorithmic}
\end{algorithm}

Compared with~\eqref{eq:fast-km-unit-step}, the update carries the error
\begin{equation}\label{eq:perturbed-recurrence}
 \eta_k=\frac{k+\alpha/2}{k+\alpha}\,e_k-\frac{k}{k+\alpha}\,e_{k-1},
\end{equation}
\begin{equation}\label{eq:conditional-update-error}
 \EE[\eta_k\mid\FF_k]=-\frac{k}{k+\alpha}\,e_{k-1},
\end{equation}
which is nonzero in general, since $e_{k-1}$ is $\FF_k$-measurable. 
Under bounded variance, no method guarantees a residual smaller than
$D/K+\sigma/\sqrt{K}$ after $K$ queries on the whole problem class
\cite[Theorem~4.6]{park2022exact}, \cite[Theorem~2]{foster2019complexity},
whereas the last iterate of stochastic KM attains only
$O(\log k/k^{1/6})$ (Section~\ref{sec:km}). In stochastic optimization,
momentum averages the sampled gradients and reduces the variance
implicitly \cite{cutkosky2020momentum,gao2024polyak}.
Algorithm~\ref{alg:sfkm} is stochastic KM with momentum and one query per
iteration, which leads to the second question:
\blue{\begin{quote}
\emph{(Q2) Can the momentum reduce the variance implicitly and accelerate
the last iterate of stochastic KM under bounded variance?}
\end{quote}}

\subsection{Noise propagation under reuse}\label{sec:implicit-vr}

Set
\begin{equation}\label{eq:ak-dk}
 a_k:=\frac{k+\alpha/2}{k+\alpha},\qquad d_k:=\frac{k}{k+\alpha}.
\end{equation}

Since $1+d_k=2a_k$, the update of Algorithm~\ref{alg:sfkm} reads
$x_{k+1}=a_kx_k+a_k\widehat T_k(x_k)-d_k\widehat T_{k-1}(x_{k-1})$. The
following lemma isolates the noise that this update propagates.

\begin{lemma}[Noise propagation under reuse]\label{lem:noise}
Let Assumption~\ref{ass:oracle} hold and set
$b_k:=a_ka_{k-1}-d_k=\frac{\alpha(\alpha-2)}{4(k+\alpha)(k+\alpha-1)}$.
Define $u_1:=\frac12e_0$ and
\begin{equation}\label{eq:aux-noise}
 u_{k+1}=a_k\,u_k+a_k\,e_k-d_k\,e_{k-1},\qquad k\geq1 .
\end{equation}
Then $z_k:=x_k-u_k$ satisfies the deterministic Fast KM recursion with
errors $\delta_k:=Tx_k-Tz_k$, $\norm{\delta_k}\leq\norm{u_k}$. Moreover,
with $v_k:=u_k-a_{k-1}e_{k-1}$,
\begin{equation}\label{eq:noise-split}
 u_{k+1}=a_k\,e_k+v_{k+1},\qquad v_{k+1}=a_k\,v_k+b_k\,e_{k-1},\qquad v_1=0,
\end{equation}
and
\begin{equation}\label{eq:noise-recursion}
 \EE\norm{u_{k+1}}^2=a_k^2\,\EE\sigma_k^2+\EE\norm{v_{k+1}}^2,\qquad
 \EE\norm{v_{k+1}}^2=a_k^2\,\EE\norm{v_k}^2+b_k^2\,\EE\sigma_{k-1}^2 .
\end{equation}
\end{lemma}

\begin{proof}
Subtracting~\eqref{eq:aux-noise} from the update gives
$z_{k+1}=a_kz_k+a_k(Tz_k+\delta_k)-d_k(Tz_{k-1}+\delta_{k-1})$, and
$\norm{\delta_k}\leq\norm{x_k-z_k}$ by nonexpansiveness.
Substituting $u_k=v_k+a_{k-1}e_{k-1}$ into~\eqref{eq:aux-noise}
gives~\eqref{eq:noise-split}. The vector $v_k$ depends on
$e_0,\dots,e_{k-2}$ and is $\FF_{k-1}$-measurable, and $v_{k+1}$ is
$\FF_k$-measurable; conditional unbiasedness removes the cross terms
in~\eqref{eq:noise-recursion}.
\end{proof}

In~\eqref{eq:noise-recursion}, the stored error $e_{k-1}$ enters with the
net coefficient $b_k=O(1/k^2)$: its two occurrences in consecutive updates
cancel to leading order. The new error $e_k$ enters with the weight
$a_k\to1$. Compared with momentum-based variance reduction, where the new
noise enters with a tunable weight $\rho_k$
\cite{cutkosky2020momentum,gao2024polyak}, the reuse implementation with
fixed $\alpha$ reduces only the stored noise.

\begin{proposition}[Parameters of Fast KM]\label{prop:fixed}
Let $T=0$ and assume $\EE\sigma_k^2\geq\sigma^2>0$ for all $k$. Then the
iterates of Algorithm~\ref{alg:sfkm} satisfy
$\EE\norm{x_{k+1}-Tx_{k+1}}^2\geq\sigma^2/4$ for every $k\geq1$.
\end{proposition}

\begin{proof}
For $T=0$, the update is $x_{k+1}=a_kx_k+a_ke_k-d_ke_{k-1}$. Since $x_k$ and
$e_{k-1}$ are $\FF_k$-measurable, conditional unbiasedness gives
$\EE\norm{x_{k+1}}^2=\EE\norm{a_kx_k-d_ke_{k-1}}^2+a_k^2\EE\sigma_k^2
\geq\sigma^2/4$, because $a_k\geq\frac12$.
\end{proof}

\subsection{Horizon-dependent parameters}\label{sec:horizon}

With parameters that vary with $k$, momentum does accelerate the last
iterate. Consider the inertial KM iteration
\begin{equation}\label{eq:ikm}
 y_k=x_k+\alpha_k(x_k-x_{k-1}),\quad z_k=x_k+\beta_k(x_k-x_{k-1}),\quad
 x_{k+1}=(1-\lambda_k)y_k+\lambda_k\widehat T_k(z_k),
\end{equation}
which contains Fast KM as a particular choice of parameters
\cite[Appendix~A]{yang2026ikm}. For $K\geq2$ and
$0<a\leq\frac{K\log2}{\log K}-1$, set
$\omega_k:=\log\bigl(1+K^{a-(a+1)k/K}\bigr)$,
$r_k:=2^{\omega_{k-1}/\omega_k}-1$,
$u'_k:=\frac{3r_k}{2(1+r_k)(2+r_k)}$, and
\begin{equation}\label{eq:ikm-schedule}
 \beta_1=0,\quad \beta_k=\frac{4}{3r_{k-1}},\qquad
 \lambda_k=\frac{Kr_k\,\omega_{k-1}}{4(1+1/a)\,\omega_0}\,u'_k,\qquad
 \alpha_k=\frac{u'_k-\lambda_k}{1-\lambda_k}\,\beta_k .
\end{equation}

\begin{theorem}[{Varying parameters \cite[Theorem~1]{yang2026ikm}}]\label{thm:ikm}
Let Assumption~\ref{ass:operator} hold, let the errors be conditionally
unbiased with $\EE[\norm{e_k}^2\mid\FF_k]\leq\sigma^2$, and generate the
iterates by~\eqref{eq:ikm} with~\eqref{eq:ikm-schedule}. Then, for an
absolute constant $C>0$,
\[
 \Bigl(\EE\norm{x_K-Tx_K}^2\Bigr)^{1/2}
 \leq C\Bigl(\frac{1+1/a}{K}\,D+(1+a)\frac{\sigma\log K}{\sqrt{K}}\Bigr).
\]
\end{theorem}

\begin{corollary}\label{cor:ikm}
For $a=\Theta(1)$,
$\EE\norm{x_K-Tx_K}\lesssim\dfrac{D}{K}+\dfrac{\sigma\log K}{\sqrt{K}}$.
\end{corollary}

\begin{remark}[\red{Answer to (Q2)}]\label{rem:answer-q2}
\begin{enumerate}
 \item \emph{Mechanism.} Momentum reduces the variance implicitly: reuse
 cancels the stored noise (Lemma~\ref{lem:noise}), and parameters varying
 with $k$ lower the weight of the new noise (Theorem~\ref{thm:ikm}).
 \item \emph{Parameters must scale with the horizon.} With the parameters of Fast KM, the new noise
 keeps the weight $a_k\to1$, and the residual does not vanish
 (Proposition~\ref{prop:fixed}). \red{The available acceleration uses
 parameters that depend on the horizon $K$ and bounds the iterate at step
 $K$ only.}
 \item \emph{Trade-off.} For exact evaluations, Fast KM improves KM
 uniformly. Under bounded variance, the parameter $a$ trades the
 deterministic factor $1+1/a$ against the noise factor $1+a$, and the
 noise term $\sigma/\sqrt{K}$ dominates $D/K$.
\end{enumerate}
\end{remark}

\section{Fast KM under decaying noise}\label{sec:decaying}

Under bounded variance, no method guarantees a residual smaller than
$\sigma/\sqrt{N}$ after $N$ queries \cite[Theorem~2]{foster2019complexity}.
Since Corollary~\ref{cor:ikm} is already within a logarithmic factor of
this order, the rate $o(1/k)$ requires $\sigma/\sqrt{N_k}=o(1/k)$, that is,
$N_k\gg k^2$ samples in the first $k$ steps. The noise itself has to decay.
\blue{\begin{quote}
\emph{(Q3) How fast must the variance $\sigma_k^2$ decay for the reuse
implementation of stochastic Fast KM to attain
$\EE\norm{x_k-Tx_k}=o(1/k)$?}
\end{quote}}
With mini-batches $m_k\asymp k^q$, the lower bound requires only $q>1$.

\blue{For the almost-sure counterpart, the objective is to identify noise
conditions under which
\begin{equation}\label{eq:target-rate}
 k\norm{x_k-Tx_k}\longrightarrow0\qquad\text{almost surely},
\end{equation}
and, for a $\Fix T$-valued random variable $x_\infty$,
\begin{equation}\label{eq:target-iterates}
 x_k\to x_\infty\qquad\text{almost surely}.
\end{equation}}
The candidate condition is the decaying-noise condition of
\cite[Theorem~3.1]{attouch2026ravine}: $\sum_ks_k^2t_k^2\sigma_k^2<\infty$
or $\sum_ks_kt_k\sigma_k<\infty$ almost surely, where $s_k$ is the stepsize and
$t_k:=1+\sum_{i\geq k}\prod_{j=k}^{i}\alpha_j$ for the extrapolation
coefficients $\alpha_k$. For Algorithm~\ref{alg:sfkm}, $s_k\equiv1$
and $\alpha_k=d_k$, so $t_k=\frac{k+\alpha-1}{\alpha-1}$ and the condition reads
\begin{equation}\label{eq:n6-fast-km}
 \text{(N6-$\ell^2$)}\ \sum_k(k+1)^2\sigma_k^2<\infty
 \qquad\text{or}\qquad
 \text{(N6-$\ell^1$)}\ \sum_k(k+1)\sigma_k<\infty
\end{equation}
almost surely. With mini-batches $m_k\asymp k^q$ and
$\sigma_k^2=\sigma^2/m_k$, (N6-$\ell^2$) holds if and only if $q>3$, and
(N6-$\ell^1$) if and only if $q>4$.
