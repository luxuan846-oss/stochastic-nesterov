% in $\RR^d$ from stochastic oracle evaluations. Many single-loop stochastic

% momentum. Under bounded variance, the last iterate of stochastic KM attains

\section{Introduction}\label{sec:introduction}

We consider the fixed-point problem
\begin{equation}\label{eq:problem}
 \text{find }x^\star\in\HH\quad\text{such that}\quad Tx^\star=x^\star,
 \tag{FP}
\end{equation}
where $T:\HH\to\HH$ is a nonexpansive operator, that is,
$\norm{Tx-Ty}\leq\norm{x-y}$ for all $x,y\in\HH$, whose set of fixed points
$\Fix T:=\{x\in\HH:Tx=x\}$ is nonempty. Throughout, only noisy evaluations
of $T$ are available.

For exact evaluations, the fixed-point residual of KM with a constant
relaxation parameter satisfies $\norm{x_k-Tx_k}=o(1/\sqrt{k})$
\cite{bot2023fast}. The Fast KM method of Bo\c{t} and Nguyen
\cite[Algorithm~2.1]{bot2023fast} improves this to
\begin{equation}\label{eq:intro-fast-km-rate}
 \norm{x_k-Tx_k}=o(1/k),\qquad x_k\to x^\star\in\Fix T,
\end{equation}
for a momentum parameter $\alpha>2$. 

With unit stepsize, the Fast KM update reads
\begin{equation}\label{eq:fast-km-unit-step}
 x_{k+1}=x_k-\frac{\alpha}{2(k+\alpha)}\bigl(x_k-Tx_k\bigr)
 +\frac{k}{k+\alpha}\bigl(Tx_k-Tx_{k-1}\bigr),\qquad k\geq1,
\end{equation}
and uses the operator values at $x_k$ and $x_{k-1}$. With noisy
evaluations, $Tx_k$ is replaced by an oracle response
$\widehat T_k(x_k)=Tx_k+e_k$, where the subscript $k$ indicates the sample
drawn at iteration $k$. For the difference $Tx_k-Tx_{k-1}$ there are two
implementations:
\begin{itemize}
 \item \emph{Re-query with the same sample:}
 $\widehat T_k(x_k)-\widehat T_k(x_{k-1})$. Both values use the sample of
 iteration $k$. This requires two queries per iteration and an oracle that
 can evaluate one sample at two points \cite[Assumption~2.3]{lee2026fixed}.
 If the noise is Lipschitz in mean square with constant $L_\Delta$
 \cite[Assumption~2.4]{lee2026fixed}, the error of the difference is at
 most $L_\Delta\norm{x_k-x_{k-1}}$ in root mean square; see
 Appendix~\ref{app:requery}.
 Variance-reduced methods build on this access
 \cite{trandinh2024vrfkm,trandinh2025vfosa,lee2026fixed}.
 \item \emph{Reuse:}
 $\widehat T_k(x_k)-\widehat T_{k-1}(x_{k-1})$. The second value is the
 response stored at the preceding step, so each iteration queries the
 oracle once, as in the deterministic method. The error of the difference
 is $e_k-e_{k-1}$, which involves two independent samples and does not
 shrink as $x_k-x_{k-1}\to0$.
\end{itemize}
\blue{We thus study the reuse implementation.} Table~\ref{tab:rates} raises three questions:
\blue{
\begin{enumerate}
 \item \emph{(Q1) Last iterate of stochastic KM.} Under bounded variance, the best
 available last-iterate bound is $O(\log k/k^{1/6})$, whereas a randomly
 selected iterate attains $O(k^{-1/4})$. Does the last iterate necessarily
 lose this rate? (Section~\ref{sec:km})
 \item \emph{(Q2) Implicit variance reduction in Fast KM.} Under bounded variance, does the momentum of Algorithm~\ref{alg:sfkm} with fixed $\alpha>2$ reduce the variance implicitly, so that its last iterate improves on that of stochastic KM? (Section~\ref{sec:algorithm})
 \item \emph{(Q3) Fast KM under decaying noise.} For $\sigma_k^2\asymp k^{-q}$, which decay of the variance enables Algorithm~\ref{alg:sfkm} to attain $k\norm{x_k-Tx_k}\to0$ and $x_k\to x_\infty$ almost surely? (Section~\ref{sec:decaying})
\end{enumerate}}

\section{Stochastic Oracle Setting}\label{sec:problem}

% The iterate accuracy is measured by the residual $\norm{x-Tx}$. With $V:=\Id-T$, nonexpansiveness of $T$ is equivalent to
% \begin{equation}\label{eq:cocoercivity}
%  \ip{Vx-Vy}{x-y}\geq\frac12\norm{Vx-Vy}^{2},\qquad x,y\in\HH.
% \end{equation}

Let $(\Omega,\FF,\PP)$ be a probability space and $(\FF_k)_{k\geq0}$ a
filtration, where $\FF_k$ contains the information available before query
$k$. At an $\FF_k$-measurable point $x_k$, the oracle returns
\begin{equation}\label{eq:oracle}
 \widehat T_k(x_k)=Tx_k+e_k
\end{equation}
with an $\FF_{k+1}$-measurable error $e_k$.

\begin{assumption}[Oracle]\label{ass:oracle}
For every $k\geq0$,
\begin{equation}\label{eq:h1}
 \EE[e_k\mid\FF_k]=0\qquad\text{and}\qquad
 \sigma_k^2:=\EE\bigl[\norm{e_k}^2\bigm|\FF_k\bigr]
 \ \text{with}\ \EE\sigma_k^2<\infty.
 \tag{H1}
\end{equation}
\end{assumption}

\section{The last iterate of stochastic KM}\label{sec:km}
\blue{In this section, we review existing convergence results for stochastic
Krasnosel'ski\u{i}--Mann iterations in \cite{bravo2024stochastic}.}

% For Problem~\eqref{eq:problem}, the Krasnosel'ski\u{i}--Mann (KM)
% iteration reads
% \begin{equation}\label{eq:km}
%  x_{k+1}=(1-\lambda_{k+1})x_k+\lambda_{k+1}Tx_k,\qquad k\geq0,
% \end{equation}
% with $x_0\in\HH$ and parameters $\lambda_k\in(0,1)$. 

% With a constant $\lambda_k\equiv\lambda$, the iterates converge to a point of $\Fix T$, and the rate is
% \[
% \norm{x_k-Tx_k}=o(1/\sqrt{k}).
% \]

The stochastic KM iteration is
\begin{equation}\label{eq:skm}
 x_{k+1}=(1-\lambda_{k+1})x_k+\lambda_{k+1}\widehat T_k(x_k),\qquad
 \widehat T_k(x_k)=Tx_k+e_k .
\end{equation}
Under bounded variance, a randomly selected iterate generated by
\eqref{eq:skm} attains the order $k^{-1/4}$
\cite[Corollary~5.4]{bravo2024stochastic}, \cite{cortild2026stochastic}.
This leads to the following question:
\blue{
\begin{quote}
\emph{(Q1) For a general nonexpansive $T$ and bounded variance, can the
last iterate of stochastic KM attain the order $k^{-1/4}$?}
\end{quote}}

Let $D:=\dist(x_0,\Fix T)$ and set
$\tau_k:=\sum_{j=1}^{k}\lambda_j(1-\lambda_j)$,
$\varphi(y):=\min\{1,1/\sqrt{\pi y}\}$ and
\[
 \nu_k^2:=\sum_{j=1}^{k}\bigl(w_j^k\bigr)^2\,\EE\sigma_{j-1}^2,
 \qquad w_j^k:=\lambda_j\prod_{i=j+1}^{k}(1-\lambda_i),
\]
We recall the conditions used for the last-iterate bound
in \cite{bravo2024stochastic}.
\begin{assumption}\label{ass:h4}
Either $T$ has bounded range and $\bar\kappa\geq\sup_{x}\norm{Tx-x_0}$, or
$S:=\sum_{j\geq2}\lambda_j\nu_{j-1}<\infty$ and $\bar\kappa\geq2D+S$.
\end{assumption}
The following theorem gives a bound on the expected fixed-point residual
of the last iterate of stochastic KM
\cite[Theorem~2.10]{bravo2024stochastic}.
\begin{theorem}\label{thm:skm}
Let $T$ be nonexpansive with $\Fix T\neq\emptyset$, and let Assumptions~\ref{ass:oracle} and~\ref{ass:h4} hold.
Then, for every $k\geq1$,
\begin{equation}\label{eq:skm-bound}
 \EE\norm{x_k-Tx_k}\leq\bar\kappa\,\varphi(\tau_k)
 +2\sum_{j=2}^{k}\lambda_j\,\varphi(\tau_k-\tau_j)\,\nu_{j-1}
 +4\nu_k .
\end{equation}
\end{theorem}

We now consider the special case of Assumption~\ref{ass:oracle} in which
the second moments are uniformly bounded.

\blue{
\begin{assumption}[Bounded variance]\label{ass:bvar}
There exists a finite constant $\sigma\geq0$ such that
$\sup_{k\geq0}\EE\sigma_k^2\leq\sigma^2$.
\end{assumption}}

\begin{corollary}{\cite[Theorem~4.4]{bravo2024stochastic}}\label{cor:skm}
Under the assumptions of Theorem~\ref{thm:skm} and
Assumption~\ref{ass:bvar}, let $\lambda_k=(k+1)^{-a}$ with
$a\in(\frac12,1)$. Then $\tau_k\asymp k^{1-a}$ and it holds that
\[
 \EE\norm{x_k-Tx_k}\lesssim
 \underbrace{\frac{1}{\sqrt{\tau_k}}}_{\text{deterministic}}
 +\underbrace{\sigma\sqrt{k}\,\lambda_k}_{\text{noise}}
 =\frac{1}{k^{(1-a)/2}}+\frac{\sigma}{k^{a-1/2}}.
\]
Moreover, it holds that 
$\EE\norm{x_k-Tx_k}=O(\log k/k^{1/6})$ with $a=\frac23$.
\end{corollary}

\paragraph{\red{Answer to (Q1).}}\label{par:answer-q1}
\red{In the literature examined here, no: for a general nonexpansive $T$,
\[
 \EE\norm{x_k-Tx_k}=O\Bigl(\frac{\log k}{k^{1/6}}\Bigr)
 \quad\text{(last iterate)},\qquad
 \EE\norm{x_R-Tx_R}=O\Bigl(\frac{1}{k^{1/4}}\Bigr)
 \quad\text{(random iterate)}.
\]}
\red{The limitations are as follows:}
\begin{enumerate}
 \item  For exact evaluations,
 \[
  \norm{x_{k+1}-Tx_{k+1}}\leq\norm{x_k-Tx_k}
  \quad\Longrightarrow\quad
  \norm{x_k-Tx_k}^2\leq\frac{D^2}{\tau_{k+1}} ;
 \]
 with noise, $\EE\norm{x_k-Tx_k}^2$ need not be monotone, and the summed
 bound applies only to $x_R$ \cite[Remark~5.3]{bravo2024stochastic}.
 \item \emph{Order of the last iterate:} in~\eqref{eq:skm-bound},
 $1/\sqrt{\tau_k}\asymp\sqrt{k}\,\lambda_k\asymp k^{-1/6}$ at $a=\frac23$
 (Corollary~\ref{cor:skm}).
\end{enumerate}

\section{Implicit variance reduction in Fast KM}\label{sec:algorithm}
\subsection{The reuse implementation of Stochastic Fast KM}\label{sec:iteration}

The Fast KM method \cite[Algorithm~2.1]{bot2023fast} also has a stepsize
$0<s\leq1$. Its iteration with stepsize $s$ applied to $T$ coincides with
\eqref{eq:fast-km-unit-step} applied to $M:=(1-s)\Id+sT$, which is
nonexpansive with $\Fix M=\Fix T$, and an oracle for $T$ with error $e_k$
yields an oracle for $M$ with error $se_k$. We therefore take $s=1$ without
loss of generality.

Set $\rho_k:=\frac{\alpha}{k+\alpha}$, so that $\frac{\alpha}{2(k+\alpha)}=\frac{\rho_k}{2}$ and $\frac{k}{k+\alpha}=1-\rho_k$ in~\eqref{eq:fast-km-unit-step}. Algorithm~\ref{alg:sfkm} replaces $Tx_k$ by the fresh response $\widehat T_k(x_k)=Tx_k+e_k$ and $Tx_{k-1}$ by the stored response $\widehat T_{k-1}(x_{k-1})=Tx_{k-1}+e_{k-1}$.

\begin{algorithm}[t]
\color{black}
\captionsetup{font={color=black}}
\caption{Stochastic Fast KM with reuse}\label{alg:sfkm}
\begin{algorithmic}[1]
\STATE \textbf{Input:} $x_0,x_1\in\HH$, $\alpha>2$.
\STATE Query $\widehat T_0(x_0)$ and store it.
\FOR{$k=1,2,\dots$}
 \STATE Query $\widehat T_k(x_k)=Tx_k+e_k$.
 \STATE $x_{k+1}=x_k-\frac{\rho_k}{2}\bigl(x_k-\widehat T_k(x_k)\bigr)
 +(1-\rho_k)\bigl(\widehat T_k(x_k)-\widehat T_{k-1}(x_{k-1})\bigr)$.
 \STATE Store $\widehat T_k(x_k)$.
\ENDFOR
\end{algorithmic}
\end{algorithm}

In stochastic optimization, momentum averages the sampled gradients and
reduces the variance implicitly \cite{cutkosky2020momentum,gao2024polyak}.
This leads to the second question:
\blue{\begin{quote}
\emph{(Q2) Under bounded variance, does the momentum of Algorithm~\ref{alg:sfkm} with fixed $\alpha>2$ reduce the variance implicitly, so that its last iterate improves on that of stochastic KM?}
\end{quote}}

\subsection{Noise propagation under reuse}\label{sec:implicit-vr}

The update of Algorithm~\ref{alg:sfkm} splits into the exact step and the errors:
\[
 x_{k+1}=\underbrace{x_k-\frac{\rho_k}{2}\bigl(x_k-Tx_k\bigr)+(1-\rho_k)\bigl(Tx_k-Tx_{k-1}\bigr)}_{\text{exact Fast KM step}}
 +\underbrace{\frac{\rho_k}{2}}_{\text{relaxation}\,\to\,0}e_k
 +\underbrace{(1-\rho_k)}_{\text{momentum}\,\to\,1}\bigl(e_k-e_{k-1}\bigr).
\]
Compared with~\eqref{eq:fast-km-unit-step}, the update thus carries the error
\begin{equation}\label{eq:perturbed-recurrence}
 \eta_k=\Bigl(1-\frac{\rho_k}{2}\Bigr)e_k-(1-\rho_k)\,e_{k-1}.
\end{equation}
The stored error $e_{k-1}$ enters $\eta_{k-1}$ and $\eta_k$ with coefficients of opposite signs, which nearly cancel (Lemma~\ref{lem:noise}). The new error $e_k$ enters with the weight
\[
\underbrace{\frac{\rho_k}{2}}_{\text{relaxation}}+\underbrace{(1-\rho_k)}_{\text{momentum}}=1-\frac{\rho_k}{2}\ \to\ 1,
\]
so the momentum term offsets the damping of the relaxation step. For $T=0$, this gives a lower bound on the residual.

\begin{proposition}\label{prop:fixed} Let $T=0$ and assume $\mathbb{E}\sigma_k^2\geq\sigma^2>0$ for all $k$. Then the iterates of Algorithm~\ref{alg:sfkm} satisfy, for every $k\geq1$,

\[
\mathbb{E}\lVert x_{k+1}-Tx_{k+1}\rVert^2\geq\frac{\sigma^2}{4}.
\]

\end{proposition}

\begin{proof} For $T=0$, the update reads $x_{k+1}=\bigl(1-\frac{\rho_k}{2}\bigr)x_k+\bigl(1-\frac{\rho_k}{2}\bigr)e_k-(1-\rho_k)e_{k-1}$. Since $x_k$ and $e_{k-1}$ are $\mathcal{F}_k$-measurable, conditional unbiasedness gives $\mathbb{E}\lVert x_{k+1}\rVert^2\geq\bigl(1-\frac{\rho_k}{2}\bigr)^2\mathbb{E}\sigma_k^2\geq\frac{\sigma^2}{4}$, because $\rho_k\leq1$. \end{proof}

\paragraph{\red{Answer to (Q2).}}\label{rem:answer-q2}
\red{For fixed $\alpha>2$: no. The residual of Algorithm~\ref{alg:sfkm}
need not even vanish in mean square under bounded variance
(Proposition~\ref{prop:fixed}).}

\paragraph{The perturbation argument.} Existing analyses of Fast KM concern exact evaluations or summable perturbations \cite{yang2026ikm}, and bounded variance does not make $\eta_k$ in~\eqref{eq:perturbed-recurrence} summable. Following \cite[Section~2.1]{bravo2024stochastic}, we reformulate the iteration with an auxiliary sequence that absorbs the errors. Let $u_0:=0$, $u_1:=\frac12e_0$, and define

\begin{equation}\label{eq:aux-noise}
u_{k+1}=\Bigl(1-\frac{\rho_k}{2}\Bigr)u_k+\Bigl(1-\frac{\rho_k}{2}\Bigr)e_k-(1-\rho_k)\,e_{k-1},\qquad k\geq1.
\end{equation}

Subtracting this from the update and denoting $z_k:=x_k-u_k$, we get

\[
z_{k+1}=z_k-\frac{\rho_k}{2}\bigl(z_k-Tz_k-\delta_k\bigr)+(1-\rho_k)\bigl(Tz_k+\delta_k-Tz_{k-1}-\delta_{k-1}\bigr),
\]

which is the deterministic Fast KM iteration with errors $\delta_k:=Tx_k-Tz_k$ satisfying $\lVert\delta_k\rVert\leq\lVert x_k-z_k\rVert=\lVert u_k\rVert$.

For stochastic KM, the corresponding process is an average of the errors and converges to zero under the noise and stepsize conditions in that analysis, so the effective errors $\delta_k$ become small \cite[Section~2.1]{bravo2024stochastic}. The next lemma shows that $u_k$ is not an average.

\begin{lemma}[Noise propagation under reuse]\label{lem:noise} Let Assumption~\ref{ass:oracle} hold. Set $v_k:=u_k-\bigl(1-\frac{\rho_{k-1}}{2}\bigr)e_{k-1}$ and $b_k:=\bigl(1-\frac{\rho_k}{2}\bigr)\bigl(1-\frac{\rho_{k-1}}{2}\bigr)-(1-\rho_k)$. Then $v_1=0$, $0\leq b_k\leq\frac{\rho_k\rho_{k-1}}{4}$, and for all $k\geq1$,

\begin{equation}\label{eq:noise-split}
u_{k+1}=\Bigl(1-\frac{\rho_k}{2}\Bigr)e_k+v_{k+1},\qquad v_{k+1}=\Bigl(1-\frac{\rho_k}{2}\Bigr)v_k+b_k\,e_{k-1},
\end{equation}

\begin{equation}\label{eq:noise-recursion}
\begin{split}
\mathbb{E}\lVert u_{k+1}\rVert^2&=\Bigl(1-\frac{\rho_k}{2}\Bigr)^2\mathbb{E}\sigma_k^2+\mathbb{E}\lVert v_{k+1}\rVert^2,\\
\mathbb{E}\lVert v_{k+1}\rVert^2&=\Bigl(1-\frac{\rho_k}{2}\Bigr)^2\mathbb{E}\lVert v_k\rVert^2+b_k^2\,\mathbb{E}\sigma_{k-1}^2.
\end{split}
\end{equation}

\end{lemma}

\begin{proof} Since $\rho_0=1$, $v_1=u_1-\frac12e_0=0$. Substituting $u_k=v_k+\bigl(1-\frac{\rho_{k-1}}{2}\bigr)e_{k-1}$ into the recursion of $u_k$ gives the recursion of $v_k$. A direct computation gives $b_k=\frac{\alpha(\alpha-2)}{4(k+\alpha)(k+\alpha-1)}$, which lies in $\bigl[0,\frac{\rho_k\rho_{k-1}}{4}\bigr]$. The vector $v_k$ depends only on $e_0,\dots,e_{k-2}$, and $v_{k+1}$ is $\mathcal{F}_k$-measurable. Conditional unbiasedness therefore removes the cross terms. \end{proof}

\begin{remark}\label{rem:reuse-vr} By Lemma~\ref{lem:noise}, the stored error enters $u_{k+1}$ with the net coefficient $b_k=O(1/k^2)$ and the new error with the weight $1-\frac{\rho_k}{2}\to1$. Hence, if $\inf_k\mathbb E\sigma_k^2>0$, then $u_k$ does not converge to zero in mean square, and the estimate $\lVert\delta_k\rVert\leq\lVert u_k\rVert$ does not make the effective errors small. Bravo and Cominetti observe that stochastic KM carries implicitly a form of variance reduction that operates as the iterates evolve \cite[Remark~2.3]{bravo2024stochastic}. For Algorithm~\ref{alg:sfkm}, this variance reduction acts only on the stored errors. Momentum-based methods control the contribution of new stochastic information through tunable parameters \cite{cutkosky2020momentum,gao2024polyak}.
\end{remark}


\section{Fast KM under decaying noise}\label{sec:decaying}

Proposition~\ref{prop:fixed} shows that the noise has to decay. Its proof
gives, for $T=0$,
\[
 \mathbb E\lVert x_{k+1}-Tx_{k+1}\rVert^2
 \geq\Bigl(1-\frac{\rho_k}{2}\Bigr)^2\mathbb E\sigma_k^2
 \geq\frac14\mathbb E\sigma_k^2,
\]
so the root-mean-square rate
$(\mathbb E\lVert x_k-Tx_k\rVert^2)^{1/2}=o(1/k)$ for $T=0$ requires
\begin{equation}\label{eq:ms-necessary}
 k^2\,\mathbb E\sigma_k^2\longrightarrow0.
\end{equation}
A candidate sufficient condition for almost-sure convergence and rates is
the decaying-noise condition $\sum_ks_k^2t_k^2\sigma_k^2<\infty$ almost
surely of \cite[Theorem~3.1]{attouch2026ravine}, where $s_k$ is the
stepsize and $t_k:=1+\sum_{i\geq k}\prod_{j=k}^{i}\alpha_j$ for the
extrapolation coefficients $\alpha_k$. For Algorithm~\ref{alg:sfkm},
$s_k\equiv1$ and $\alpha_k=1-\rho_k$, so $t_k=\frac{k+\alpha-1}{\alpha-1}$
and the condition reads
\begin{equation}\label{eq:n6-fast-km}
 \sum_k(k+1)^2\sigma_k^2<\infty\qquad\text{almost surely}.
 \tag{N6-$\ell^2$}
\end{equation}
Theorem~3.1 of \cite{attouch2026ravine} is stated for the Ravine method,
and its applicability to Algorithm~\ref{alg:sfkm} remains to be
established.

Both conditions concern the weighted variance $k^2\sigma_k^2$:
\eqref{eq:ms-necessary} requires it to vanish in expectation, and
\eqref{eq:n6-fast-km} requires it to be summable. For
$\sigma_k^2\asymp k^{-q}$, they correspond to $q>2$ and $q>3$,
respectively. The former is necessary for the mean-square rate; the
latter is a candidate condition for the almost-sure rate. This leads to the
third question:
\blue{\begin{quote}
\emph{(Q3) For $\sigma_k^2\asymp k^{-q}$, which decay ensures that
Algorithm~\ref{alg:sfkm} attains $k\norm{x_k-Tx_k}\to0$ and
$x_k\to x_\infty$ almost surely, for a $\Fix T$-valued random variable
$x_\infty$? In particular, does the almost-sure threshold lie at $q>2$, as
for the mean-square rate, or at $q>3$?}
\end{quote}}

\section{Re-query with the same sample}\label{app:requery}

This appendix compares the two implementations of
Section~\ref{sec:introduction} in the notation of
Section~\ref{sec:algorithm}. Let $T(\cdot;\xi)$ be a sampled operator with
$\EE[T(y;\xi_k)]=Ty$ for every $y\in\HH$, where the sample $\xi_k$ is
independent of $\FF_k$. Set $\widehat T_k(y):=T(y;\xi_k)$ and
\[
 e_k(y):=\widehat T_k(y)-Ty,\qquad y\in\HH.
\]
The error of Section~\ref{sec:algorithm} is $e_k=e_k(x_k)$. Since $x_k$ and
$x_{k-1}$ are $\FF_k$-measurable,
$\EE[e_k(x_k)\mid\FF_k]=\EE[e_k(x_{k-1})\mid\FF_k]=0$.

\paragraph{Two updates.} With $\rho_k=\frac{\alpha}{k+\alpha}$, the
re-query implementation evaluates $\widehat T_k$ at $x_k$ and at $x_{k-1}$:
\begin{equation}\label{eq:requery-update}
 x_{k+1}=x_k-\frac{\rho_k}{2}\bigl(x_k-\widehat T_k(x_k)\bigr)
 +(1-\rho_k)\bigl(\widehat T_k(x_k)-\widehat T_k(x_{k-1})\bigr).
\end{equation}
This requires two queries per iteration and same-sample access
\cite[Assumption~2.3]{lee2026fixed}. Algorithm~\ref{alg:sfkm} replaces
$\widehat T_k(x_{k-1})$ by the stored response $\widehat T_{k-1}(x_{k-1})$
and uses each sample once.

\paragraph{Errors.} Compared with~\eqref{eq:fast-km-unit-step}, the two
updates carry the errors
\begin{equation}\label{eq:requery-error}
 \eta_k^{\mathrm{rq}}=\frac{\rho_k}{2}\,e_k
 +(1-\rho_k)\bigl(e_k-e_k(x_{k-1})\bigr),
 \qquad
 \eta_k=\frac{\rho_k}{2}\,e_k+(1-\rho_k)\bigl(e_k-e_{k-1}\bigr).
\end{equation}
They differ only in the error at $x_{k-1}$: the re-query implementation
uses the error $e_k(x_{k-1})$ of the current sample, and the reuse
implementation uses the error $e_{k-1}=e_{k-1}(x_{k-1})$ of the preceding
sample.

Assume the noise is Lipschitz in mean square, as in
\cite[Assumption~2.4]{lee2026fixed}:
$\EE\norm{e_k(x)-e_k(y)}^2\leq L_\Delta^2\norm{x-y}^2$ for all
$x,y\in\HH$. Since $x_k$ and $x_{k-1}$ are $\FF_k$-measurable and $\xi_k$ is
independent of $\FF_k$,
\[
 \EE\bigl[\norm{e_k(x_k)-e_k(x_{k-1})}^2\bigm|\FF_k\bigr]
 \leq L_\Delta^2\norm{x_k-x_{k-1}}^2.
\]
In the reuse implementation, $e_{k-1}$ is $\FF_k$-measurable and
$\EE[e_k\mid\FF_k]=0$, so
\[
 \EE\norm{e_k-e_{k-1}}^2=\EE\sigma_k^2+\EE\sigma_{k-1}^2,
\]
which does not decrease as $x_k-x_{k-1}\to0$. Hence, in
$\eta_k^{\mathrm{rq}}$ the new error $e_k$ enters outside the difference
only with the weight $\frac{\rho_k}{2}$, whereas in $\eta_k$ it enters with
the weight $1-\frac{\rho_k}{2}$ (Lemma~\ref{lem:noise}).

\paragraph{Variance-reduced estimators.} The STORM-type recursion of VRAF
\cite[(VRAF)]{lee2026fixed}, written for $T$ in the present indexing, is
\[
 d_k=\widehat T_k(x_k)+(1-\beta_k)\bigl(d_{k-1}-\widehat T_k(x_{k-1})\bigr),
 \qquad\beta_k\in(0,1].
\]
Its error $\varepsilon_k:=d_k-Tx_k$ satisfies
\[
 \varepsilon_k=(1-\beta_k)\,\varepsilon_{k-1}+\beta_k\,e_k
 +(1-\beta_k)\bigl(e_k-e_k(x_{k-1})\bigr).
\]
The new error enters with the weight $\beta_k$, and the remaining term is
the same-sample difference controlled above. The loopless SVRG estimator of
\cite[(7)]{trandinh2024vrfkm} for Fast KM likewise evaluates the same
mini-batch at $x_k$ and at $x_{k-1}$. Both constructions rely on
same-sample access, which the reuse implementation does not assume.
